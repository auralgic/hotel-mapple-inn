export default function App() {
  const [cart, setCart] = useState({});
  const [activeCategory, setActiveCategory] = useState(MENU_CATEGORIES[0].id);
  const [orderState, setOrderState] = useState('menu'); 
  
  // Checkout Form State
  const [customerName, setCustomerName] = useState('');
  const [customerPhone, setCustomerPhone] = useState('');
  const [tableNumber, setTableNumber] = useState('');
  const [transactionId, setTransactionId] = useState('');

  const categoryRefs = useRef({});
  const categoryScrollRef = useRef(null);

  const handleScroll = () => {
    const scrollPosition = window.scrollY + 220; 
    let currentCategory = MENU_CATEGORIES[0].id;
    
    for (let i = MENU_CATEGORIES.length - 1; i >= 0; i--) {
      const category = MENU_CATEGORIES[i];
      const element = categoryRefs.current[category.id];
      if (element && element.offsetTop <= scrollPosition) {
        currentCategory = category.id;
        break;
      }
    }
    setActiveCategory(currentCategory);

    // Auto-scroll the top category bar to keep active item visible
    const activeBtn = document.getElementById(`cat-btn-${currentCategory}`);
    if (activeBtn && categoryScrollRef.current) {
        activeBtn.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'center' });
    }
  };

  useEffect(() => {
    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, []);

  const scrollToCategory = (categoryId) => {
    const element = categoryRefs.current[categoryId];
    if (element) {
      window.scrollTo({
        top: element.offsetTop - 180,
        behavior: 'smooth'
      });
    }
  };

  const getCartId = (itemId, variantName) => variantName ? `${itemId}-${variantName}` : itemId;

  const updateCart = (item, delta, variant = null) => {
    const cartId = getCartId(item.id, variant?.name);
    
    setCart(prev => {
      const currentQty = prev[cartId]?.quantity || 0;
      const newQty = Math.max(0, currentQty + delta);
      
      if (newQty === 0) {
        const newCart = { ...prev };
        delete newCart[cartId];
        return newCart;
      }
      
      return {
        ...prev,
        [cartId]: {
          ...item,
          variantName: variant?.name,
          price: variant?.price || item.price,
          quantity: newQty
        }
      };
    });
  };

  const cartTotalAmount = Object.values(cart).reduce((total, item) => total + (item.price * item.quantity), 0);
  const cartTotalItems = Object.values(cart).reduce((total, item) => total + item.quantity, 0);

  // Updated UPI ID logic
  const upiId = "8295547753@ptaxis"; 
  const upiString = `upi://pay?pa=${upiId}&pn=Hotel%20Mapple%20Inn&am=${cartTotalAmount}&cu=INR`;
  const qrCodeUrl = `https://api.qrserver.com/v1/create-qr-code/?size=250x250&data=${encodeURIComponent(upiString)}`;

  const placeOrder = (e) => {
    e.preventDefault();
    if (!customerName || !customerPhone || !tableNumber || !transactionId) {
      alert("Please fill all required fields and enter the UTR / Transaction ID.");
      return;
    }
    setOrderState('success');
  };

  const resetOrder = () => {
    setCart({});
    setCustomerName('');
    setCustomerPhone('');
    setTableNumber('');
    setTransactionId('');
    setOrderState('menu');
  };

  if (orderState === 'success') {
    return (
      <div className="min-h-screen bg-gray-50 flex flex-col items-center justify-center p-4 font-sans">
        <div className="bg-white p-8 rounded-3xl shadow-xl max-w-md w-full text-center border border-gray-100">
          <CheckCircle2 className="w-20 h-20 text-green-500 mx-auto mb-4" />
          <h2 className="text-2xl font-bold text-gray-800 mb-2">Order Confirmed!</h2>
          <p className="text-gray-600 mb-6">
            Thank you, <strong>{customerName}</strong>. Your meal for <strong>Table {tableNumber}</strong> is being prepared.
          </p>
          
          <div className="bg-amber-50 text-amber-900 p-4 rounded-xl mb-6 text-left shadow-sm">
            <p className="font-semibold mb-3 border-b border-amber-200 pb-2">Order Summary</p>
            <ul className="text-sm space-y-2 mb-3">
              {Object.values(cart).map((item, idx) => (
                <li key={idx} className="flex justify-between items-start">
                  <span className="flex-1 pr-2">{item.quantity}x {item.name} <span className="text-xs text-amber-700 block">{item.variantName ? `(${item.variantName})` : ''}</span></span>
                  <span className="font-medium">₹{item.price * item.quantity}</span>
                </li>
              ))}
            </ul>
            <div className="border-t border-amber-200 pt-2 font-bold flex justify-between text-base mb-2">
              <span>Total Amount</span>
              <span>₹{cartTotalAmount}</span>
            </div>
            <div className="bg-white p-3 rounded-lg border border-amber-200 text-sm mt-3">
               <p className="text-gray-500 mb-1">Payment Method:</p>
               <p className="font-bold text-gray-800">PhonePe / UPI</p>
               <p className="text-xs text-gray-500 mt-1 break-all">Txn ID: {transactionId}</p>
            </div>
          </div>
          <button 
            onClick={resetOrder}
            className="w-full bg-amber-600 text-white font-bold py-3.5 rounded-xl shadow-md hover:bg-amber-700 transition"
          >
            Start New Order
          </button>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-gray-50 font-sans pb-28">
      {/* HEADER & VISUAL CATEGORY NAV */}
      <header className="bg-white shadow-[0_4px_20px_rgba(0,0,0,0.03)] sticky top-0 z-40 pb-2 rounded-b-2xl">
        <div className="max-w-3xl mx-auto px-4 py-4 flex justify-between items-center">
          <div>
            <h1 className="text-xl font-extrabold text-gray-900 flex items-center gap-2 tracking-tight">
               🍁 Hotel Mapple Inn
            </h1>
            <p className="text-[10px] font-bold text-amber-600 tracking-widest uppercase mt-1">Premium Pure Veg</p>
          </div>
          <button 
            onClick={() => cartTotalItems > 0 && setOrderState('checkout')}
            className={`relative p-2.5 rounded-full transition-all ${cartTotalItems > 0 ? 'bg-amber-100 text-amber-700 shadow-inner' : 'bg-gray-100 text-gray-400'}`}
          >
            <ShoppingCart className="w-5 h-5" />
            {cartTotalItems > 0 && (
              <span className="absolute -top-1.5 -right-1.5 bg-red-500 text-white text-[10px] font-bold w-5 h-5 flex items-center justify-center rounded-full border-2 border-white shadow-sm">
                {cartTotalItems}
              </span>
            )}
          </button>
        </div>
        
        {/* VISUAL CATEGORY CAROUSEL */}
        {orderState === 'menu' && (
          <div 
            ref={categoryScrollRef}
            className="max-w-3xl mx-auto px-4 overflow-x-auto no-scrollbar flex gap-4 pt-1 pb-2"
          >
            {MENU_CATEGORIES.map(category => {
              const isActive = activeCategory === category.id;
              return (
                <button
                  key={category.id}
                  id={`cat-btn-${category.id}`}
                  onClick={() => scrollToCategory(category.id)}
                  className="flex flex-col items-center gap-1.5 min-w-[72px] transition-transform duration-200"
                >
                  <div className={`w-[68px] h-[68px] rounded-full overflow-hidden p-1 transition-all duration-300 ${isActive ? 'bg-gradient-to-tr from-amber-400 to-orange-500 shadow-md scale-105' : 'bg-transparent'}`}>
                    <img 
                      src={category.image} 
                      alt={category.name}
                      className="w-full h-full object-cover rounded-full border-2 border-white"
                      loading="lazy"
                    />
                  </div>
                  <span className={`text-[11px] font-semibold text-center leading-tight transition-colors ${isActive ? 'text-amber-600' : 'text-gray-500'}`}>
                    {category.name}
                  </span>
                </button>
              );
            })}
          </div>
        )}
      </header>

      {/* MAIN CONTENT AREA */}
      <main className="max-w-3xl mx-auto px-4 py-6">
        {orderState === 'menu' ? (
          <div className="space-y-10">
            {MENU_CATEGORIES.map(category => {
              const items = MENU_ITEMS.filter(item => item.category === category.id);
              if (items.length === 0) return null;
              
              return (
                <section key={category.id} ref={el => categoryRefs.current[category.id] = el} className="scroll-mt-48">
                  <h2 className="text-xl font-bold text-gray-800 mb-4 flex items-center gap-2 pl-1">
                    {category.name}
                  </h2>
                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                    {items.map(item => (
                      <div key={item.id} className="bg-white p-3 rounded-2xl shadow-sm border border-gray-100 flex gap-4 overflow-hidden relative">
                        
                        {/* Information Section */}
                        <div className="flex-1 flex flex-col justify-between py-1">
                          <div>
                            <div className="flex justify-between items-start mb-1 gap-2">
                              <h3 className="font-bold text-gray-800 text-[15px] leading-tight">{item.name}</h3>
                              <span className="shrink-0 bg-green-50 text-green-600 p-0.5 rounded border border-green-200/50">
                                  <UtensilsCrossed className="w-3 h-3" />
                              </span>
                            </div>
                            {item.desc && <p className="text-xs text-gray-500 mb-3 line-clamp-2 leading-relaxed pr-2">{item.desc}</p>}
                          </div>
                          
                          {/* Price & Add Controls */}
                          {item.variants ? (
                            <div className="mt-2 flex flex-col gap-2">
                              {item.variants.map((variant, idx) => {
                                const cartId = getCartId(item.id, variant.name);
                                const qty = cart[cartId]?.quantity || 0;
                                return (
                                  <div key={idx} className="flex justify-between items-center bg-gray-50/80 p-2 rounded-xl">
                                    <span className="text-sm font-bold text-gray-800">₹{variant.price} <span className="text-[10px] text-gray-500 font-medium uppercase tracking-wide">({variant.name})</span></span>
                                    {qty === 0 ? (
                                      <button onClick={() => updateCart(item, 1, variant)} className="text-xs bg-amber-50 text-amber-700 font-bold px-4 py-1.5 rounded-lg border border-amber-200 shadow-sm hover:bg-amber-100 transition">ADD</button>
                                    ) : (
                                      <div className="flex items-center gap-2 bg-amber-600 text-white rounded-lg px-2 py-1 shadow-sm">
                                        <button onClick={() => updateCart(item, -1, variant)} className="p-0.5 hover:bg-amber-700 rounded"><Minus className="w-3.5 h-3.5" /></button>
                                        <span className="text-xs font-bold w-4 text-center">{qty}</span>
                                        <button onClick={() => updateCart(item, 1, variant)} className="p-0.5 hover:bg-amber-700 rounded"><Plus className="w-3.5 h-3.5" /></button>
                                      </div>
                                    )}
                                  </div>
                                )
                              })}
                            </div>
                          ) : (
                            <div className="mt-2 flex justify-between items-end">
                              <span className="text-[17px] font-black text-gray-800">₹{item.price}</span>
                              {(() => {
                                const qty = cart[item.id]?.quantity || 0;
                                return qty === 0 ? (
                                  <button onClick={() => updateCart(item, 1)} className="text-sm bg-amber-50 text-amber-700 font-bold px-6 py-1.5 rounded-xl border border-amber-200 shadow-sm hover:bg-amber-100 transition">
                                    ADD
                                  </button>
                                ) : (
                                  <div className="flex items-center gap-3 bg-amber-600 text-white rounded-xl px-3 py-1.5 shadow-sm">
                                    <button onClick={() => updateCart(item, -1)} className="p-0.5 hover:bg-amber-700 rounded"><Minus className="w-4 h-4" /></button>
                                    <span className="text-sm font-bold w-4 text-center">{qty}</span>
                                    <button onClick={() => updateCart(item, 1)} className="p-0.5 hover:bg-amber-700 rounded"><Plus className="w-4 h-4" /></button>
                                  </div>
                                );
                              })()}
                            </div>
                          )}
                        </div>

                        {/* Individual Item Image Section */}
                        <div className="w-32 h-32 shrink-0 relative rounded-2xl overflow-hidden shadow-sm self-center">
                          <img 
                            src={item.image} 
                            alt={item.name}
                            className="w-full h-full object-cover transition-transform duration-700 hover:scale-110"
                            loading="lazy"
                          />
                          <div className="absolute inset-0 bg-gradient-to-t from-black/20 to-transparent pointer-events-none"></div>
                        </div>

                      </div>
                    ))}
                  </div>
                </section>
              );
            })}
          </div>
        ) : (
          /* CHECKOUT SCREEN */
          <div className="animate-in fade-in slide-in-from-bottom-4 duration-300 max-w-lg mx-auto">
            <button 
              onClick={() => setOrderState('menu')}
              className="flex items-center gap-2 text-gray-600 mb-6 hover:text-gray-900 font-medium bg-white px-4 py-2.5 rounded-xl shadow-sm border border-gray-100 w-fit"
            >
              <ArrowLeft className="w-4 h-4" /> Back to Menu
            </button>
            
            <h2 className="text-2xl font-extrabold text-gray-900 mb-4 tracking-tight">Complete Order</h2>
            
            {/* Cart Summary */}
            <div className="bg-white rounded-3xl shadow-sm border border-gray-100 p-5 mb-6">
              <h3 className="font-bold text-gray-800 mb-4 pb-3 border-b border-gray-100 flex items-center gap-2">
                Order Items
              </h3>
              {Object.values(cart).map((item, idx) => (
                 <div key={idx} className="flex justify-between items-center py-2.5 border-b border-gray-50 last:border-0 last:pb-0">
                   <div className="flex-1 pr-4">
                     <h4 className="font-bold text-gray-800 text-sm leading-tight">
                       {item.name} {item.variantName && <span className="text-amber-600 text-[10px] uppercase font-bold block mt-0.5">{item.variantName}</span>}
                     </h4>
                     <div className="text-xs font-medium text-gray-500 mt-1">₹{item.price} x {item.quantity}</div>
                   </div>
                   <div className="w-20 text-right font-black text-sm text-gray-800 pl-2">
                     ₹{item.price * item.quantity}
                   </div>
                 </div>
              ))}
              <div className="mt-4 pt-4 border-t-2 border-dashed border-gray-200 flex justify-between items-center">
                <span className="font-bold text-gray-600">Total Bill</span>
                <span className="text-xl font-black text-amber-600">₹{cartTotalAmount}</span>
              </div>
            </div>

            {/* Checkout Form */}
            {cartTotalItems > 0 && (
              <form onSubmit={placeOrder} className="mb-24 space-y-6">
                
                {/* Section 1: Customer Details */}
                <div className="bg-white rounded-3xl shadow-sm border border-gray-100 p-5">
                  <h3 className="font-bold text-gray-800 mb-4 flex items-center gap-2"><User className="w-4 h-4 text-amber-500"/> Personal Details</h3>
                  <div className="space-y-4">
                    <div>
                      <label className="block text-xs font-bold text-gray-600 mb-1.5 uppercase tracking-wider">Full Name <span className="text-red-500">*</span></label>
                      <input 
                        type="text" 
                        required
                        value={customerName}
                        onChange={(e) => setCustomerName(e.target.value)}
                        placeholder="Enter your name" 
                        className="w-full border-2 border-gray-100 rounded-xl px-4 py-3 focus:outline-none focus:border-amber-400 focus:ring-4 focus:ring-amber-50 transition-all font-medium text-sm"
                      />
                    </div>
                    <div>
                      <label className="block text-xs font-bold text-gray-600 mb-1.5 uppercase tracking-wider">Contact Number <span className="text-red-500">*</span></label>
                      <div className="flex relative">
                        <span className="absolute left-4 top-1/2 -translate-y-1/2 text-gray-400 text-sm"><Phone className="w-4 h-4"/></span>
                        <input 
                          type="tel" 
                          required
                          pattern="[0-9]{10}"
                          title="Please enter a valid 10-digit mobile number"
                          value={customerPhone}
                          onChange={(e) => setCustomerPhone(e.target.value)}
                          placeholder="10-digit mobile number" 
                          className="w-full border-2 border-gray-100 rounded-xl pl-10 pr-4 py-3 focus:outline-none focus:border-amber-400 focus:ring-4 focus:ring-amber-50 transition-all font-medium text-sm"
                        />
                      </div>
                    </div>
                    <div>
                      <label className="block text-xs font-bold text-gray-600 mb-1.5 uppercase tracking-wider">Table Number <span className="text-red-500">*</span></label>
                      <div className="flex relative">
                        <span className="absolute left-4 top-1/2 -translate-y-1/2 text-gray-400 text-sm"><MapPin className="w-4 h-4"/></span>
                        <input 
                          type="number" 
                          required
                          value={tableNumber}
                          onChange={(e) => setTableNumber(e.target.value)}
                          placeholder="e.g. 5" 
                          className="w-full border-2 border-gray-100 rounded-xl pl-10 pr-4 py-3 focus:outline-none focus:border-amber-400 focus:ring-4 focus:ring-amber-50 transition-all font-medium text-sm"
                        />
                      </div>
                    </div>
                  </div>
                </div>

                {/* Section 2: UPI Payment Only */}
                <div className="bg-white rounded-3xl shadow-sm border border-gray-100 p-5">
                  <h3 className="font-bold text-gray-800 mb-4 flex items-center gap-2"><QrCode className="w-4 h-4 text-indigo-500"/> Scan & Pay</h3>
                  
                  <div className="flex flex-col items-center animate-in fade-in duration-300 pt-2">
                    <p className="text-sm font-bold text-indigo-900 mb-4 text-center">Scan to pay exactly <strong className="text-xl bg-indigo-100 px-2 py-1 rounded">₹{cartTotalAmount}</strong></p>
                    
                    <div className="p-4 bg-white border border-indigo-100 rounded-3xl shadow-sm mb-5">
                      <img 
                        src={qrCodeUrl} 
                        alt="UPI Payment QR Code" 
                        className="w-52 h-52 rounded-xl"
                      />
                    </div>
                    
                    <div className="w-full bg-indigo-50/50 p-4 rounded-2xl">
                      <label className="block text-xs font-bold text-indigo-900 mb-2 uppercase tracking-wider">Enter Transaction ID (UTR) <span className="text-red-500">*</span></label>
                      <input 
                        type="text" 
                        required
                        value={transactionId}
                        onChange={(e) => setTransactionId(e.target.value)}
                        placeholder="e.g. 3125xxxxxxxx" 
                        className="w-full border-2 border-indigo-100 rounded-xl px-4 py-3 focus:outline-none focus:border-indigo-500 focus:ring-4 focus:ring-indigo-100 transition-all font-mono font-bold text-sm bg-white"
                      />
                      <p className="text-[10px] font-medium text-indigo-400 mt-2 text-center">Please pay and enter the 12-digit UTR number to confirm.</p>
                    </div>
                  </div>
                </div>

                <button 
                  type="submit"
                  className="w-full font-bold text-lg py-4 rounded-2xl shadow-lg transition-all flex items-center justify-center gap-2 transform active:scale-[0.98] bg-amber-600 text-white hover:bg-amber-700 shadow-amber-600/30"
                >
                  Confirm & Place Order <ChevronRight className="w-5 h-5" />
                </button>
              </form>
            )}
          </div>
        )}
      </main>

      {/* FLOATING CART BAR */}
      {orderState === 'menu' && cartTotalItems > 0 && (
        <div className="fixed bottom-0 left-0 right-0 p-4 pb-6 bg-white/80 backdrop-blur-xl border-t border-gray-200/50 shadow-[0_-10px_40px_rgba(0,0,0,0.08)] z-50 animate-in slide-in-from-bottom-full duration-300">
          <div className="max-w-3xl mx-auto flex items-center justify-between">
            <div>
              <p className="text-[11px] text-amber-700 font-black uppercase tracking-widest mb-0.5">{cartTotalItems} Item{cartTotalItems > 1 ? 's' : ''} Added</p>
              <p className="text-2xl font-black text-gray-900 tracking-tight">₹{cartTotalAmount} <span className="text-[10px] text-gray-500 font-bold uppercase tracking-wider ml-1">+ taxes</span></p>
            </div>
            <button 
              onClick={() => setOrderState('checkout')}
              className="bg-gray-900 text-white px-8 py-3.5 rounded-2xl font-bold flex items-center gap-2 shadow-xl shadow-gray-900/20 hover:bg-black transition transform active:scale-95"
            >
              Checkout <ChevronRight className="w-5 h-5" />
            </button>
          </div>
        </div>
      )}
      
      {/* GLOBAL STYLES */}
      <style dangerouslySetInnerHTML={{__html: `
        .no-scrollbar::-webkit-scrollbar { display: none; }
        .no-scrollbar { -ms-overflow-style: none; scrollbar-width: none; }
        body { -webkit-tap-highlight-color: transparent; scroll-behavior: smooth; }
      `}} />
    </div>
  );
}
